   387→const DISK_IDENTIFY = `# 3.A — Identifikasi disk SSD (OS) & HDD (DATA)
   388→# SSD 128GB biasanya sudah jadi disk OS (sda). HDD 500GB untuk DATA.
   389→
   390→# 1) List semua block device
   391→lsblk -o NAME,SIZE,TYPE,MOUNTPOINT,FSTYPE,MODEL
   392→
   393→# 2) Cek disk terpasang
   394→sudo fdisk -l | grep -E '^Disk /dev/(sd|nvme)'
   395→
   396→# 3) Identifikasi SSD vs HDD berdasarkan rotasi (0=SSD, 1=HDD)
   397→for dev in /sys/block/sd* /sys/block/nvme*; do
   398→  [ -d "$dev" ] || continue
   399→  name=$(basename "$dev")
   400→  rotational=$(cat "$dev/queue/rotational" 2>/dev/null)
   401→  size=$(lsblk -dn -o SIZE "/dev/$name" 2>/dev/null)
   402→  [ "$rotational" = "0" ] && echo "$name (SSD, $size)" || echo "$name (HDD, $size)"
   403→done
   404→
   405→# CONTOH OUTPUT:
   406→# sda (SSD, 128G)   ← disk OS, jangan diformat!
   407→# sdb (HDD, 500G)   ← disk DATA, akan di-mount ke /mnt/data`
   408→
   409→const DISK_FORMAT = `# 3.B — Format HDD 500GB sebagai ext4 (JIKA belum diformat)
   410→# HATI-HATI: pastikan /dev/sdb BENAR-BENAR HDD data kosong, bukan disk OS!
   411→# Cek ulang dengan: sudo blkid /dev/sdb1 — harus TIDAK ADA output jika kosong.
   412→
   413→# 1) Partisi HDD (single partition, GPT)
   414→sudo parted -s /dev/sdb mklabel gpt
   415→sudo parted -s /dev/sdb mkpart primary ext4 0% 100%
   416→
   417→# 2) Format ext4 (pilihan terbaik untuk Linux file server — journaling, ACL support)
   418→sudo mkfs.ext4 -L lawu-data -m 1 -T largefile4 /dev/sdb1
   419→#   -L lawu-data  = label disk (memudah identifikasi)
   420→#   -m 1          = reserved 1% untuk root (default 5% terlalu boros di 500GB)
   421→#   -T largefile4 = optimasi untuk file besar (ebook PDF/EPUB, index Tantivy)
   422→
   423→# 3) Verifikasi format
   424→sudo blkid /dev/sdb1
   425→# Harus cetak: TYPE="ext4" LABEL="lawu-data"`

   990→async fn search_handler(
   991→    State(state): State<AppState>,
   992→    Json(req): Json<SearchRequest>,
   993→) -> Result<Json<SearchResponse>, StatusCode> {
   994→    let searcher = state.reader.searcher();
   995→    let f = &state.schema_fields;
   996→
   997→    // QueryParser: cari di title + body (BM25 default)
   998→    let query_parser = QueryParser::for_index(
   999→        &state.index,
  1000→        vec![f.title, f.body],
  1001→    );
  1002→    let query = query_parser
  1003→        .parse_query(&req.query)
  1004→        .map_err(|_| StatusCode::BAD_REQUEST)?;
  1005→
  1006→    let limit = req.limit.unwrap_or(60).min(200);
  1007→    let top_docs = searcher
  1008→        .search(&query, &TopDocs::with_limit(limit))
  1009→        .map_err(|_| StatusCode::INTERNAL_SERVER_ERROR)?;
  1010→
  1011→    let hits: Vec<SearchHit> = top_docs
  1012→        .into_iter()
  1013→        .map(|(score, doc_addr)| {
  1014→            let doc: tantivy::TantivyDocument = searcher.doc(doc_addr).unwrap_or_default();
  1015→            SearchHit {
  1016→                title: doc.get_first(f.title).and_then(|v| v.as_str()).unwrap_or("").to_string(),
  1017→                path: doc.get_first(f.path).and_then(|v| v.as_str()).unwrap_or("").to_string(),
  1018→                file_type: doc.get_first(f.file_type).and_then(|v| v.as_str()).unwrap_or("").to_string(),
  1019→                category: doc.get_first(f.category).and_then(|v| v.as_str()).unwrap_or("").to_string(),
  1020→                score,
  1021→            }
  1022→        })
  1023→        .collect();
  1024→
  1025→    Ok(Json(SearchResponse { total: hits.len(), results: hits }))
  1026→}
