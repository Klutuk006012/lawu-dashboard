    18→def extract_constants(source: str) -> dict:
    19→    """Extract semua `const NAME = `...`` dari source TypeScript."""
    20→    constants = {}
    21→    pattern = re.compile(r'^const (\w+) = `', re.MULTILINE)
    22→    matches = list(pattern.finditer(source))
    23→    for m in matches:
    24→        name = m.group(1)
    25→        start = m.end()
    26→        # Cari closing backtick yang TIDAK didahului \\
    27→        idx = start
    28→        while idx < len(source):
    29→            bt_pos = source.find('`', idx)
    30→            if bt_pos == -1:
    31→                break
    32→            # Cek apakah backtick didahului \\
    33→            bs_count = 0
    34→            j = bt_pos - 1
    35→            while j >= 0 and source[j] == '\\':
    36→                bs_count += 1
    37→                j -= 1
    38→            # Jika jumlah backslash genap, backtick tidak di-escape
    39→            if bs_count % 2 == 0:
    40→                raw = source[start:bt_pos]
    41→                content = raw.replace('\\`', '`').replace('\\\\', '\\').replace('\\$', '$')
    42→                constants[name] = content
    43→                break
    44→            idx = bt_pos + 1
    45→    return constants
    46→
