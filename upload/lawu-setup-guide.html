Mount & Format HDD/SSD + Automount Reboot

Identifikasi SSD (OS) vs HDD (DATA), format ulang HDD 500GB ext4 (WAJIB jika HDD ikut terformat saat install ulang OS), AUTOMOUNT via /etc/fstab dengan nofail (server tetap boot walau HDD error) + test reboot, tuning SSD (TRIM, I/O scheduler).

!
Hati-hati memilih disk! · HDD ikut terformat saat install ulang?
Pastikan /dev/sdb BENAR-BENAR HDD data, bukan disk OS — nama device BISA berubah setelah install ulang, selalu verifikasi dengan 3.A. Jika HDD ikut terformat installer: data lama hilang, lanjut format ulang di 3.B lalu automount di 3.C, download ulang model GGUF di Langkah 11. WAJIB test reboot (3.C langkah 7-8) — tanpa fstab benar, /mnt/data kosong tiap reboot dan semua service gagal start.
3.A-identify-disks.sh sh 
# 3.A — Identifikasi disk SSD (OS) & HDD (DATA)
# SSD 128GB biasanya sudah jadi disk OS (sda). HDD 500GB untuk DATA.
# CATATAN (kasus install ulang OS + HDD ikut terformat): HDD akan muncul
# TANPA MOUNTPOINT dan mungkin TANPA FSTYPE — itulah HDD data kita.
# Data lama (model GGUF, ebook) sudah HILANG — mulai dari nol.

# 1) List semua block device
lsblk -o NAME,SIZE,TYPE,MOUNTPOINT,FSTYPE,MODEL

# 2) Cek disk terpasang
sudo fdisk -l | grep -E '^Disk /dev/(sd|nvme)'

# 3) Identifikasi SSD vs HDD berdasarkan rotasi (0=SSD, 1=HDD)
for dev in /sys/block/sd* /sys/block/nvme*; do
  [ -d "$dev" ] || continue
  name=$(basename "$dev")
  rotational=$(cat "$dev/queue/rotational" 2>/dev/null)
  size=$(lsblk -dn -o SIZE "/dev/$name" 2>/dev/null)
  [ "$rotational" = "0" ] && echo "$name (SSD, $size)" || echo "$name (HDD, $size)"
done

# CONTOH OUTPUT (setelah install ulang OS — HDD ikut terformat):
# sda (SSD, 128G)   ← disk OS Debian 13 baru, JANGAN DIFORMAT ULANG!
# sdb (HDD, 500G)   ← disk DATA, kosong (terformat installer), siap dipersiapkan ulang
#
# PENTING: nama device BISA BERUBAH setelah install ulang!
# HDD yang dulu sdb bisa sekarang terbaca sda. Selalu identifikasi ulang —
# JANGAN asal copy-paste nama device dari catatan lama.
3.B-format-hdd-ext4.sh sh 
# 3.B — Format ulang HDD 500GB sebagai ext4
# Kasus: HDD ikut terformat saat install ulang OS → data lama hilang,
# format ulang di sini 100% AMAN (tidak ada yang bisa hilang lagi).
# HATI-HATI TETAP: pastikan /dev/sdb BENAR-BENAR HDD data, bukan disk OS!
# Cek dulu: sudo blkid /dev/sdb* — boleh kosong (blank) atau sisa installer.

# 1) Partisi HDD (single partition, GPT — timpa partisi sisa installer)
sudo parted -s /dev/sdb mklabel gpt
sudo parted -s /dev/sdb mkpart primary ext4 0% 100%

# 2) Format ext4 (pilihan terbaik untuk Linux file server — journaling, ACL support)
sudo mkfs.ext4 -L lawu-data -m 1 -T largefile4 /dev/sdb1
#   -L lawu-data  = label disk (memudah identifikasi)
#   -m 1          = reserved 1% untuk root (default 5% terlalu boros di 500GB)
#   -T largefile4 = optimasi untuk file besar (ebook PDF/EPUB, index Tantivy)

# 3) Verifikasi format
sudo blkid /dev/sdb1
# Harus cetak: TYPE="ext4" LABEL="lawu-data"
#
# INFO: model GGUF Qwen2.5-Coder-7B Q3_K_M (3.55GB) yang lama HILANG —
# download ulang di Langkah 11 (opsi 1: wget dari bartowski, paling cepat).
3.C-fstab-automount.sh sh 
# 3.C — AUTOMOUNT saat reboot via /etc/fstab (WAJIB — mount manual HILANG saat reboot!)
# Tanpa fstab: tiap reboot server, /mnt/data kosong → llama-server & Axum GAGAL start.

# 1) Dapat UUID partisi HDD (UUID permanen — tidak berubah saat reboot,
#    tidak seperti /dev/sdb yang bisa pindah jadi sda)
UUID=$(sudo blkid -s UUID -o value /dev/sdb1)
echo "UUID HDD: $UUID"

# 2) Backup fstab lama
sudo cp /etc/fstab /etc/fstab.bak.$(date +%Y%m%d)

# 3) Tambah entry automount
echo "UUID=$UUID  /mnt/data  ext4  defaults,noatime,nofail,x-systemd.device-timeout=10s  0  2" \
  | sudo tee -a /etc/fstab

#   defaults      = rw, suid, dev, exec, auto, nouser, async
#   noatime       = jangan update access time (hemat write HDD, lebih cepat)
#   nofail        = KRUSIAL! server TETAP BOOT walau HDD gagal/cabut.
#                   Tanpa nofail: HDD error = server stuck di boot = DOWN TOTAL!

use tantivy::schema::{Schema, TEXT, STORED, FAST};

pub fn build_schema() -> Schema {
    // Skema multi-field — identik dengan kontrak dashboard:
    // title, body, path, file_type, category, pages, size_bytes
    let mut schema = Schema::builder();

    // title — STORED untuk di-return di hasil pencarian, TEXT untuk BM25
    schema.add_text_field("title", TEXT | STORED);

    // body — TEXT only (tidak di-store, hemat disk; snippet di-generate saat query)
    schema.add_text_field("body", TEXT);

    // path — unique per dokumen, STORED untuk lookup file asli
    schema.add_text_field("path", TEXT | STORED);

    // file_type — PDF | EPUB | DOCX (filter TermQuery presisi)
    schema.add_text_field("file_type", TEXT);

    // category — misal "Agama Islam", "Programming", "AI/ML"
    schema.add_text_field("category", TEXT);

    // pages — jumlah halaman (FAST untuk sort/filter numerik)
    schema.add_u64_field("pages", FAST | STORED);

    // size_bytes — ukuran file asli (FAST untuk sort)
    schema.add_u64_field("size_bytes", FAST | STORED);

    // NOTE: indexed_at (date field) di-add nanti saat butuh sort by timestamp.
    // Tantivy 0.24 add_date_field butuh DateOptions type annotation — skip untuk now
    // agar schema.rs pasti compile. Search by title/body BM25 sudah cukup untuk MVP.

    schema.build()
}

// Compatible dengan Axum 0.8 + Tantivy 0.24 + Rust 1.99 (2026)
use axum::{
    extract::State,
    http::StatusCode,
    routing::{get, post},
    Json, Router,
};
use serde::{Deserialize, Serialize};
use std::sync::Arc;
use tantivy::collector::TopDocs;
use tantivy::query::QueryParser;
use tantivy::schema::Value;
use tantivy::{Index, IndexReader, ReloadPolicy};

// Module declaration — schema.rs harus ada di src/schema.rs (Langkah 9)
mod schema;

#[derive(Clone)]
struct AppState {
    index: Arc<Index>,
    reader: Arc<IndexReader>,
    schema_fields: Arc<SchemaFields>,
}

struct SchemaFields {
    title: tantivy::schema::Field,
    body: tantivy::schema::Field,
    path: tantivy::schema::Field,
    file_type: tantivy::schema::Field,
    category: tantivy::schema::Field,
}

#[derive(Deserialize)]
struct SearchRequest {
    query: String,
    file_type: Option<String>,
    category: Option<String>,
    limit: Option<usize>,
}

#[derive(Serialize)]
struct SearchHit {
    title: String,
    path: String,
    file_type: String,
    category: String,
    score: f32,
}

#[derive(Serialize)]
struct SearchResponse {
    results: Vec<SearchHit>,
    total: usize,
}

#[tokio::main]
async fn main() -> anyhow::Result<()> {
    tracing_subscriber::fmt().with_env_filter("info").init();

    // 1) Buka index Tantivy di HDD, atau buat baru jika belum ada
    let index_path = "/mnt/data/tantivy_index";
    let index = Index::open_in_dir(index_path).unwrap_or_else(|_| {
        // Index belum ada — buat baru dengan schema
        std::fs::create_dir_all(index_path).expect("buat dir tantivy_index gagal");
        let schema = schema::build_schema();
        Index::create_in_dir(index_path, schema).expect("create tantivy index gagal")
    });

    // 2) Reader dengan auto-reload (indexer & searcher di proses yang sama)
    let reader = index
        .reader_builder()
        .reload_policy(ReloadPolicy::OnCommitWithDelay)
        .try_into()?;

    // 3) Cache field handles (lebih cepat daripada get_field di setiap request)
    let schema = index.schema();
    let schema_fields = Arc::new(SchemaFields {
        title: schema.get_field("title").expect("field title tidak ada"),
        body: schema.get_field("body").expect("field body tidak ada"),
        path: schema.get_field("path").expect("field path tidak ada"),
        file_type: schema.get_field("file_type").expect("field file_type tidak ada"),
        category: schema.get_field("category").expect("field category tidak ada"),
    });

    let state = AppState {
        index: Arc::new(index),
        reader: Arc::new(reader),
        schema_fields,
    };

    // 4) Router Axum 0.8 — bind 127.0.0.1:8081 (localhost only, via Nginx reverse proxy)
    let app = Router::new()
        .route("/api/stats", get(stats_handler))
        .route("/api/search", post(search_handler))
        .route("/api/telegram/test", post(telegram_test_handler))
        .with_state(state);

    let listener = tokio::net::TcpListener::bind("127.0.0.1:8081").await?;
    tracing::info!("lawu-server berjalan di http://127.0.0.1:8081");
    axum::serve(listener, app).await?;
    Ok(())
}

async fn search_handler(
    State(state): State<AppState>,
    Json(req): Json<SearchRequest>,
) -> Result<Json<SearchResponse>, StatusCode> {
    let searcher = state.reader.searcher();
    let f = &state.schema_fields;

    // QueryParser: cari di title + body (BM25 default)
    let query_parser = QueryParser::for_index(
        &state.index,
        vec![f.title, f.body],
    );
    let query = query_parser
        .parse_query(&req.query)
        .map_err(|_| StatusCode::BAD_REQUEST)?;

    let limit = req.limit.unwrap_or(60).min(200);
    let top_docs = searcher
        .search(&query, &TopDocs::with_limit(limit))
        .map_err(|_| StatusCode::INTERNAL_SERVER_ERROR)?;

    let hits: Vec<SearchHit> = top_docs
        .into_iter()
        .map(|(score, doc_addr)| {
            let doc: tantivy::TantivyDocument = searcher.doc(doc_addr).unwrap_or_default();
            SearchHit {
                title: doc.get_first(f.title).and_then(|v| v.as_str()).unwrap_or("").to_string(),
                path: doc.get_first(f.path).and_then(|v| v.as_str()).unwrap_or("").to_string(),
                file_type: doc.get_first(f.file_type).and_then(|v| v.as_str()).unwrap_or("").to_string(),
                category: doc.get_first(f.category).and_then(|v| v.as_str()).unwrap_or("").to_string(),
                score,
            }
        })
        .collect();

    Ok(Json(SearchResponse { total: hits.len(), results: hits }))
}

#!/usr/bin/env python3
# wiki_to_markdown.py — Parquet Wikipedia-ID → markdown per-artikel
import argparse, re, sys
from pathlib import Path
import pyarrow.parquet as pq

DEFAULT_OUT = "/mnt/data/processed_markdowns/wikipedia-id"
MAX_TEXT = 20000                      # truncate teks per artikel (hindari dokumen raksasa)
SAFE = re.compile(r'[^\w.-]+')

def sanitize(s):
    return SAFE.sub('_', s.strip())[:120].strip('_') or 'untitled'

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument('parquet', nargs='+')
    ap.add_argument('--split', default='train', help='subfolder split (train/test)')
    ap.add_argument('--out', default=DEFAULT_OUT, help='folder output markdown')
    ap.add_argument('--limit', type=int, default=0, help='batasi jumlah artikel (0=all)')
    args = ap.parse_args()

    out_dir = Path(args.out) / args.split
    out_dir.mkdir(parents=True, exist_ok=True)

    n, skipped = 0, 0
    for pqfile in args.parquet:
        pf = pq.ParquetFile(pqfile)
        for batch in pf.iter_batches(batch_size=500, columns=['docid', 'title', 'text']):
            for row in batch.to_pylist():
                if args.limit and n >= args.limit:
                    print(f"STOP di limit {args.limit}")
                    sys.exit(0)
                title = (row['title'] or '').strip()
                text = (row['text'] or '').strip()
                if len(text) < 40:            # skip stub/redirect terlalu pendek
                    skipped += 1
                    continue
                text = text[:MAX_TEXT]
                docid = (row['docid'] or sanitize(title)).strip()
                letter = sanitize(title)[:1].lower() if title else '_'
                sub = out_dir / (letter if letter.isalpha() else '_')
                sub.mkdir(exist_ok=True)
                fname = sub / f"{sanitize(title) or docid}__{sanitize(docid)}.md"
                fname.write_text(f"# {title}\n\n{text}\n", encoding='utf-8')
                n += 1
                if n % 10000 == 0:
                    print(f"  {n} artikel...", flush=True)
    print(f"SELESAI: {n} artikel ditulis, {skipped} dilewati → {out_dir}")

if __name__ == '__main__':
    main()
